\begin{frame}{Two triangles and a circle}
\begin{columns}[T,onlytextwidth]

\begin{column}{0.48\textwidth}
\begin{block}{\faFilm\ Heider \& Simmel (1944)}
A 90-second animation of simple shapes.
\begin{itemize}\tightitem
  \item viewers see \alert{agents with goals}
  \item they make moral judgements
  \item they get all this from motion alone
\end{itemize}
\end{block}
\begin{block}{\faUsers\ Humans do this easily}
\begin{itemize}\tightitem
  \item Heider: a few hours of stop-motion
  \item 10th-graders with a touch screen: dramatic animations in 15 minutes (Gordon \& Roemmele, 2014)
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{\faRobot\ Can a computer do this?}
First try: simulate rational agents.
\begin{itemize}\tightitem
  \item agents look lifelike, goals are clear
  \item but the motion is flat
  \item no suspense, surprise or irony
\end{itemize}
\end{block}
\begin{block}{\faLightbulb\ Question of the paper}
What does a storyteller do that a rational agent does not?
\end{block}
\end{column}

\end{columns}

\begin{column}{0.46\textwidth}
\begin{block}{\faBrain\ Run the planner backwards}
\begin{itemize}\tightitem
  \item for each possible goal: ``if this were the goal, would I move like that?''
  \item a goal gains belief when the moves \alert{fit} it
  \item update the guess after every move: Bayes' rule
\end{itemize}
\end{block}
\end{column}

\end{columns}

Inverse planning is not used to \emph{read} minds, but as a model of the \alert{audience}.
\end{alertblock}

\begin{frame}{Appendix: the maths behind the slides}
\small
\begin{columns}[T,onlytextwidth]

\begin{column}{0.48\textwidth}
\begin{block}{\faBrain\ Audience model}
\[
\belief{\Prob(H\mid s\to a)}\;\propto\;\Prob(s\to a\mid H)\,\Prob(H)
\]
{\footnotesize $H=\langle G_{\text{cheese}}, G_{\text{robot}}, \rho\rangle$; robot reward gets $\rho\cdot r_{\text{cheese}}$ extra: $\rho>0$ helps, $\rho<0$ hinders.}
\end{block}
\begin{block}{\faBullseye\ Script and help objective}
{\footnotesize $\sigma=(s_0,\sigma_1,\dots,\sigma_T)$, $\sigma_t=\langle a^{\text{robot}}_t, a^{\text{cheese}}_t, s_t\rangle$}
\[
f_{\text{help}}(\sigma)=\sum_{t=1}^{T}\Prob(\rho>0\mid\sigma_{1:t})
\]
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{\faRandom\ Device objectives}
\footnotesize
\[
f_{\text{twist}}=\sum_t
\begin{cases}
\Prob(\rho>0\mid\sigma_{1:t}) & t\le T/2\\
\Prob(\rho<0\mid\sigma_{1:t}) & t> T/2
\end{cases}
\]
\[
\begin{aligned}
f_{\text{irony}}=\sum_t\ &\Prob(G_c{=}g\mid\sigma_{1:t})+\Prob(\rho<0\mid\sigma_{1:t},G_c{=}g)\\
&+\Prob(\rho>0\mid\sigma_{1:t},G_c{=}p)
\end{aligned}
\]
\[
f_{\text{flashback}}(\sigma)=f_{\text{help}}\big(c(\sigma)\big)
\]
\[
\begin{aligned}
f_{\text{arc}}=\sum_t\ &\sin\!\big(\tfrac{t}{T}3\pi\big)\,\mathbb{E}\big[V^H_{\text{robot}}(s_t)\mid\sigma_{1:t}\big]\\
&-0.1\,D_{\mathrm{KL}}(H_{1:t-1}\,\|\,H_{1:t})
\end{aligned}
\]
\end{block}
\end{column}

\end{columns}
\end{frame}
